cmake_minimum_required(VERSION 3.16)
project(LikeEdit VERSION 1.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(LIKEEDIT_REQUIRE_SYSTEM_CONIO "Require installed LikeConio rather than fallback" OFF)
option(LIKEEDIT_USE_SYSTEM_CONIO "Use external LikeConio if discovered" ON)
set(LIKEEDIT_CONIO_SOURCE_DIR "" CACHE PATH "Optional path to original LikeConio source tree")

add_library(LikeEditCore STATIC src/editor.c src/syntax.c)
target_include_directories(LikeEditCore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
set_target_properties(LikeEditCore PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)

set(USE_SYSTEM_CONIO FALSE)
if(LIKEEDIT_USE_SYSTEM_CONIO)
  if(LIKEEDIT_CONIO_SOURCE_DIR)
    if(NOT EXISTS "${LIKEEDIT_CONIO_SOURCE_DIR}/CMakeLists.txt")
      message(FATAL_ERROR "LIKEEDIT_CONIO_SOURCE_DIR must point at the LikeConio project root")
    endif()
    set(CONIO_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(CONIO_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${LIKEEDIT_CONIO_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/likeconio-build" EXCLUDE_FROM_ALL)
  endif()
  if(NOT TARGET conio::conio)
    find_package(conio 1 CONFIG QUIET)
  endif()
  if(TARGET conio::conio)
    set(USE_SYSTEM_CONIO TRUE)
    set(LIKECONIO_TARGET conio::conio)
    message(STATUS "LikeEdit using LikeConio CMake target: conio::conio")
  else()
    find_path(LIKECONIO_INCLUDE_DIR NAMES conio.h)
    find_library(LIKECONIO_LIBRARY NAMES conio)
    if(LIKECONIO_INCLUDE_DIR AND LIKECONIO_LIBRARY)
      set(USE_SYSTEM_CONIO TRUE)
      message(STATUS "LikeEdit using installed LikeConio: ${LIKECONIO_LIBRARY}")
    endif()
  endif()
endif()
if(LIKEEDIT_REQUIRE_SYSTEM_CONIO AND NOT USE_SYSTEM_CONIO)
  message(FATAL_ERROR "LikeConio not found. Install CsmcWzrd/LikeConio and set CMAKE_PREFIX_PATH, or disable LIKEEDIT_REQUIRE_SYSTEM_CONIO")
endif()

add_executable(LikeEdit src/main.c)
target_link_libraries(LikeEdit PRIVATE LikeEditCore)
set_target_properties(LikeEdit PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
set(CURSES_NEED_WIDE TRUE)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(NCURSESW QUIET IMPORTED_TARGET ncursesw)
endif()
if(USE_SYSTEM_CONIO)
  target_compile_definitions(LikeEdit PRIVATE LIKEEDIT_SYSTEM_CONIO=1)
  if(LIKECONIO_TARGET)
    target_link_libraries(LikeEdit PRIVATE ${LIKECONIO_TARGET})
  else()
    target_include_directories(LikeEdit PRIVATE ${LIKECONIO_INCLUDE_DIR})
    target_link_libraries(LikeEdit PRIVATE ${LIKECONIO_LIBRARY})
    # A manually located static libconio.a does not carry its ncursesw link flags.
    if(NCURSESW_FOUND)
      target_link_libraries(LikeEdit PRIVATE PkgConfig::NCURSESW)
    else()
      find_package(Curses REQUIRED)
      target_link_libraries(LikeEdit PRIVATE ${CURSES_LIBRARIES})
    endif()
  endif()
else()
  target_sources(LikeEdit PRIVATE src/compat_conio.c)
  if(NCURSESW_FOUND)
    target_link_libraries(LikeEdit PRIVATE PkgConfig::NCURSESW)
  else()
    find_package(Curses REQUIRED)
    target_include_directories(LikeEdit PRIVATE ${CURSES_INCLUDE_DIRS})
    target_link_libraries(LikeEdit PRIVATE ${CURSES_LIBRARIES})
  endif()
  message(STATUS "LikeConio not installed; building bundled ncurses compatibility backend")
endif()

if(BUILD_TESTING)
  add_executable(LikeEditCoreTests tests/test_core.c)
  target_link_libraries(LikeEditCoreTests PRIVATE LikeEditCore)
  add_test(NAME core COMMAND LikeEditCoreTests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND AND UNIX)
    add_test(NAME terminal COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/pty_smoke.py $<TARGET_FILE:LikeEdit>)
  endif()
endif()
install(TARGETS LikeEdit RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES README.md docs/PORTING.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
